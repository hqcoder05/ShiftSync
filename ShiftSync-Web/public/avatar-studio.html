React Three Fiber Model Code
// Based on: DEV Community Tutorial
// "How to Use Three.js And React to Render a 3D Model of Your Self"
import React, { useRef, Suspense } from "react";
import { Canvas } from "@react-three/fiber";
import { OrbitControls, useGLTF } from "@react-three/drei";

export function Model(props) {
  const group = useRef();
  const { scene } = useGLTF("/models/ShiftSync_Employee.glb");
  return (
    <group ref={group} {...props}>
      <primitive object={scene} />
    </group>
  );
}

export default function App() {
  return (
    <Canvas
      camera={{ position: [0, 0, 3.4], fov: 32 }}
      style={{ backgroundColor: "#111a21" }}
    >
      <ambientLight intensity={1.2} />
      <directionalLight position={[5, 8, 5]} intensity={1.4} />
      <Suspense fallback={null}>
        <Model position={[0, -0.9, 0]} />
      </Suspense>
      <OrbitControls />
    </Canvas>
  );
}
